**Helper function for PNG embedding:**
```python
def _embed_chart(fig):
    """Convert matplotlib figure to base64 PNG and return nbformat output."""
    buf = io.BytesIO()
    fig.savefig(buf, format="png", dpi=CHART_STYLE["dpi"], bbox_inches="tight")
    buf.seek(0)
    img_b64 = base64.b64encode(buf.getvalue()).decode("utf-8")
    plt.close(fig)
    return new_output(
        output_type='display_data',
        data={'image/png': img_b64}
    )
```

**Main notebook builder function:**
```python
def build_notebook(panels, sigs, bt):
    """Build .ipynb notebook with sections A–E, charts, and sensitivity tables."""
    nb = new_notebook()
    
    # Title + metadata
    nb.cells.append(new_markdown_cell("# Options Quant — Sleeve Allocation Framework (α Engine)"))
    nb.cells.append(new_markdown_cell(f"**Latest update:** {dt.utcnow().isoformat(timespec='seconds')} UTC"))
    
    # Section A: Current State (top-of-page)
    nb.cells.append(new_markdown_cell(section_a_state(sigs)))
    
    # Chart: Signal time-series (all signals, pct rank)
    sigs_to_plot = [s for s in sigs.pct.keys() if s != 'fragility']  # Skip fragility (optional)
    fig, axes = plt.subplots(
        len(sigs_to_plot), 1,
        figsize=CHART_STYLE["figsize"],
        sharex=True
    )
    if len(sigs_to_plot) == 1:
        axes = [axes]
    
    for ax, sig in zip(axes, sigs_to_plot):
        sigs.pct[sig].plot(ax=ax, lw=CHART_STYLE["linewidth"])
        ax.axhline(0.5, color=CHART_STYLE["color_grid"], ls="--", lw=0.5)
        ax.set_ylim(0, 1)
        ax.set_title(f"{sig} (causal 5y pct rank)", fontsize=CHART_STYLE["fontsize_title"])
        ax.legend(loc="upper left", fontsize=CHART_STYLE["fontsize_legend"])
    plt.tight_layout()
    
    nb.cells.append(new_code_cell("# Chart: Signal percentile ranks over time"))
    nb.cells[-1]['outputs'] = [_embed_chart(fig)]
    
    # Section B: Sleeve Mechanics
    nb.cells.append(new_markdown_cell(section_b_mechanics()))
    
    # Section C: Bucket Returns + Holm Results
    nb.cells.append(new_markdown_cell(section_c_buckets(sigs, bt)))
    
    # Section D: Past Periods That Looked Like Now (reframed)
    nb.cells.append(new_markdown_cell(section_d_analog(sigs, bt)))
    
    # Section E: Subperiod Stability
    nb.cells.append(new_markdown_cell(section_e_subperiod(bt)))
    
    # Section G: Transaction Cost Sensitivity
    tc_table = tc_sensitivity_table(bt)
    nb.cells.append(new_markdown_cell(format_tc_grid(tc_table)))
    
    # Section H: Tail-Risk Metrics
    tr_table = tail_metrics_table(bt)
    nb.cells.append(new_markdown_cell(format_tail_metrics(tr_table)))
    
    return nb
```

    with open(NB_PATH, "w") as f:

    print(f"Notebook saved to {NB_PATH.resolve()}")

if __name__ == "__main__":
    main()
