def _fig_to_b64(fig) -> str:
    buf = io.BytesIO()
    fig.savefig(buf, format="png", dpi=110, bbox_inches="tight")
    buf.seek(0)
    b64 = base64.b64encode(buf.getvalue()).decode("utf-8")
    plt.close(fig)
    return b64
```

CSS = """

def build_html(panels, sigs, bt) -> str:
    parts = []
    parts.append(f"<!DOCTYPE html><html><head><meta charset='utf-8'>"
                 f"<title>Options Quant POC</title>"
                 f"<style>{CSS}</style></head><body>")
    parts.append("<h1>Options Quant — Sleeve Allocation Framework (α Engine)</h1>")
    parts.append(f"<p class='meta'>Generated: {datetime.utcnow().strftime('%Y-%m-%d %H:%M')} UTC · "
                 f"Data: CBOE + FRED (free sources) · "
                 f"90mny IV: synthesized (slope=0.2 approximation — see WALKTHROUGH.md)</p>")

    # Section A — leads the report
    parts.append("<h2>Section A — Current State</h2>")
    parts.append(f"<pre>{section_a_state(sigs)}</pre>")

    # Signal chart
    sig_b64 = _signal_chart(sigs)
    parts.append(f"<img src='data:image/png;base64,{sig_b64}' alt='Signal percentile ranks'>")

    # Section B
    parts.append("<h2>Section B — Sleeve Mechanics</h2>")
    parts.append(f"<pre>{section_b_mechanics()}</pre>")

    # Section C
    parts.append("<h2>Section C — Bucket Returns &amp; Holm Correction</h2>")
    parts.append(f"<pre>{section_c_buckets(sigs, bt)}</pre>")

    # Section D (reframed — Past Periods That Looked Like Now)
    parts.append("<h2>Section D — Past Periods That Looked Like Now</h2>")
    parts.append(f"<pre>{section_d_analog(sigs, bt)}</pre>")

    # Section E
    parts.append("<h2>Section E — Subperiod Stability</h2>")
    parts.append(f"<pre>{section_e_subperiod(bt)}</pre>")

    # Section G — TC Sensitivity
    parts.append("<h2>Section G — Transaction Cost Sensitivity</h2>")
    parts.append(f"<pre>{format_tc_grid(tc_sensitivity_table(bt))}</pre>")

    # Section H — Tail Risk
    parts.append("<h2>Section H — Tail-Risk Metrics</h2>")
    parts.append(f"<pre>{format_tail_metrics(tail_metrics_table(bt))}</pre>")

    parts.append("</body></html>")
    return "\n".join(parts)
```

**Entry point:**
```python
def main():
    out_dir = pathlib.Path("out")
    out_dir.mkdir(exist_ok=True)
    fname = out_dir / f"sleeve_report_{datetime.utcnow().strftime('%Y%m%d')}.html"

    print("Loading data...")
    panels = build_panels(start="2010-01-01")
    print("Building signals...")
    sigs = build_signals(panels)
    print("Running backtest...")
    bt = run_backtest(panels, sigs)
    print("Generating HTML report...")
    html = build_html(panels, sigs, bt)

    fname.write_text(html, encoding="utf-8")
    print(f"Report saved: {fname.resolve()}")

if __name__ == "__main__":
    main()
```
