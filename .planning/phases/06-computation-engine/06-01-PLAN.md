```python
"""
Pure positioning-context computation functions.

All functions are pure (no I/O, no side effects, no network calls).
compute.py orchestrates data fetching and passes data in.

Design:
    - compute_rv20(): close-to-close annualized vol from price history
    - compute_vrp(): IV30 minus RV20 — hedging cost context
    - gex_percentile(): net GEX rank vs trailing history
    - positioning_narrative(): hardcoded mechanical description of GEX sign

Methodology notes:
    - RV20 uses close-to-close log-return std × √252 (trading days).
      CBOE IV30 is already annualized. Both in percentage points.
      VRP may carry ~1-3pp systematic bias from calendar-day vs trading-day
      annualization (structural, not a bug — document, don't "fix").
    - gex_percentile requires PERCENTILE_MIN_HISTORY observations (30) before
      returning a value. Below that threshold percentile = noise.
    - positioning_narrative hardcodes exactly 2 narratives (positive/negative GEX).
      No dynamic generation. Banned words: "expect", "likely", "should", "will",
      "predict", "ahead", "upcoming". Each metric's narrative is independent.
"""
from __future__ import annotations

import numpy as np
import pandas as pd
from scipy.stats import percentileofscore

from gex import config

# ── Hardcoded narrative strings ────────────────────────────────────────────────
# Frozen mechanical descriptions. DO NOT add forward-looking language.
# Reviewed against banned-word list. Approved 2026-05-22.

_NARRATIVE_POSITIVE = (
    "Dealers are net long gamma. To remain delta-neutral they sell into rallies "
    "and buy dips — mechanically dampening intraday moves and drawing spot toward "
    "the \u03b3-flip level. This describes the hedging flow implied by current "
    "open interest; it is not a directional forecast."
)

_NARRATIVE_NEGATIVE = (
    "Dealers are net short gamma. To remain delta-neutral they buy into rallies "
    "and sell into dips — mechanically amplifying moves away from the \u03b3-flip "
    "level. This describes the hedging flow implied by current open interest; it "
    "is not a directional forecast."
)

_NARRATIVE_NONE = (
    "Insufficient data to characterise current dealer gamma positioning."
)


def compute_rv20(prices: pd.Series) -> float | None:
    """
    20-day close-to-close realized volatility, annualized by √252, in percentage points.

    Returns None if fewer than config.RV20_MIN_HISTORY price observations are
    available (produces unreliable estimate below this threshold).

    Args:
        prices: Series of daily close prices (any length), most recent last.
                NaN values are dropped before computation.

    Returns:
        Annualized RV as a float percentage (e.g., 18.5 means 18.5%), or None.

    Note: Close-to-close estimator understates realized vol during gap-heavy
    regimes. Label displays as "RV20 (c-t-c)". Yang-Zhang upgrade deferred to v3.3+.
    """
    prices = prices.dropna()
    if len(prices) < config.RV20_MIN_HISTORY:
        return None
    recent = prices.iloc[-config.RV20_MIN_HISTORY:]
    log_rets = np.log(recent / recent.shift(1)).dropna()
    return float(log_rets.std() * np.sqrt(252) * 100)


def compute_vrp(iv30: float | None, rv20: float | None) -> float | None:
    """
    Variance Risk Premium = IV30 − RV20 (both in percentage points).

    Positive VRP: implied vol > realized vol → options are expensive relative
    to recent realized moves (option sellers collecting premium).
    Negative VRP: implied vol < realized vol → options are cheap relative to
    recent realized moves.

    Returns None if either input is None (graceful degradation).

    Note: IV30 and RV20 use different annualization windows (~30 calendar days
    vs 20 trading days ≈ 28 calendar days). The ~1-3pp systematic bias is
    structural and consistent with academic convention (Carr & Wu 2009).
    """
    if iv30 is None or rv20 is None:
        return None
    return float(iv30 - rv20)


def gex_percentile(net_gex: float, history: pd.Series) -> int | None:
    """
    Percentile rank of net_gex vs trailing GEX history.

    Returns None if fewer than config.PERCENTILE_MIN_HISTORY observations exist.
    Below 30 observations, percentile is noise (same class of error as the killed
    $200M neutral floor). Returns int in range 0–100.

    Args:
        net_gex: Current session net GEX scalar (dollars).
        history: Series of historical net_gex values (NaN values dropped).
    """
    history = history.dropna()
    if len(history) < config.PERCENTILE_MIN_HISTORY:
        return None
    return int(percentileofscore(history, net_gex, kind="rank"))


def positioning_narrative(net_gex: float | None) -> str:
    """
    Hardcoded mechanical positioning narrative based on GEX sign.

    Exactly 2 narratives (positive/negative) plus a data-absent edge case.
    No dynamic generation. No compound signals. Present-tense only.

    Args:
        net_gex: Current net GEX scalar, or None if unavailable.

    Returns:
        Narrative string. Always returns str (never None).
    """
    if net_gex is None or net_gex == 0:
        return _NARRATIVE_NONE
    return _NARRATIVE_POSITIVE if net_gex > 0 else _NARRATIVE_NEGATIVE
```
