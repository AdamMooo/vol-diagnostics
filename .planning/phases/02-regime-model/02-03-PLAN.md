```python
# Extract posterior probabilities from the fitted model (predict_proba = forward-backward, NOT Viterbi)
# posteriors shape: (n_observations, 2) — column i = P(state i | observations)
posteriors = model.predict_proba(X)

# Determine which state index corresponds to Risk-On (set in cell 10)
risk_on_state_idx = 0 if label_0 == 'Risk-On' else 1

# Create regime_posteriors — canonical output for Phase 5 visualization
regime_posteriors = pd.Series(posteriors[:, risk_on_state_idx], index=bench_aligned.index)
regime_posteriors.name = 'p_risk_on'

# Posterior stats
print("=== POSTERIOR PROBABILITY: P(Risk-On) ===")
print(f"  Mean P(Risk-On):           {regime_posteriors.mean():.2%}")
print(f"  Min:                       {regime_posteriors.min():.2%}")
print(f"  Max:                       {regime_posteriors.max():.2%}")
print(f"  Days with P(Risk-On) > 0.5: {(regime_posteriors > 0.5).sum()} "
      f"({(regime_posteriors > 0.5).mean():.1%} of history)")
print(f"\nOutput variable ready: regime_posteriors ({len(regime_posteriors):,} observations)")
```
