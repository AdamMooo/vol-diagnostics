```python
from hmmlearn.hmm import GaussianHMM

# Validate input (per 02-RESEARCH.md Assumption A5)
assert bench_aligned.isnull().sum() == 0, "bench_aligned contains NaN — alignment failed"
assert len(bench_aligned) > 100, "Not enough observations for HMM fitting"

# Reshape to (n_observations, 1) — hmmlearn requires 2D input
X = bench_aligned.values.reshape(-1, 1)

# Fit 2-state Gaussian HMM (covariance_type='diag' appropriate for 1D input)
model = GaussianHMM(n_components=2, covariance_type='diag', n_iter=100, tol=1e-4, random_state=0)
model.fit(X)

# Convergence check (non-convergence = unreliable states)
print(f"Converged: {model.monitor_.converged}")
print(f"Log-likelihood: {model.score(X):.4f}")
if not model.monitor_.converged:
    print("WARNING: EM algorithm did not converge — consider increasing n_iter or loosening tol")

# Model parameters
print(f"\nTransition matrix:\n{model.transmat_}")
print(f"Mean returns per state: {model.means_.flatten()}")
print(f"Volatility per state:   {np.sqrt(model.covars_.flatten())}")

# Viterbi decode — most likely state sequence
state_sequence = model.predict(X)

# Create date-indexed Series
regime_decoded = pd.Series(state_sequence, index=bench_aligned.index)
regime_decoded.name = 'regime_state'

print(f"\nState 0 observations: {(regime_decoded == 0).sum()}")
print(f"State 1 observations: {(regime_decoded == 1).sum()}")
print(f"First 20 states: {regime_decoded.head(20).values}")
```

```python
# Split bench_aligned by decoded state (reuse compute_summary_stats from Phase 1 cell 6)
state_0_returns = bench_aligned[regime_decoded == 0]
state_1_returns = bench_aligned[regime_decoded == 1]

stats_0 = compute_summary_stats(state_0_returns, risk_free_rate=0.02)
stats_1 = compute_summary_stats(state_1_returns, risk_free_rate=0.02)

# Composite majority-vote labeling (Decision D-01)
# Higher annual return → +1 for Risk-On
# Lower annual vol → +1 for Risk-On
# Higher Sharpe ratio → +1 for Risk-On
# State winning ≥2 of 3 signals = Risk-On

def label_states_by_composite(stats_0, stats_1):
    """Assign Risk-On/Risk-Off labels via majority vote on three signals (D-01)."""
    score_0 = 0
    score_0 += 1 if stats_0['Annual Return'] > stats_1['Annual Return'] else 0
    score_0 += 1 if stats_0['Annual Volatility'] < stats_1['Annual Volatility'] else 0
    score_0 += 1 if stats_0['Sharpe Ratio'] > stats_1['Sharpe Ratio'] else 0
    label_0 = 'Risk-On' if score_0 >= 2 else 'Risk-Off'
    label_1 = 'Risk-Off' if score_0 >= 2 else 'Risk-On'
    return label_0, label_1, score_0

label_0, label_1, score_0 = label_states_by_composite(stats_0, stats_1)

# Stats display for audit (Decision D-02)
stats_df = pd.DataFrame({
    'State 0': stats_0,
    'State 1': stats_1,
}).T

# Format for readability
fmt_table = stats_df.copy()
for col in ['Annual Return', 'Annual Volatility', 'Max Drawdown']:
    fmt_table[col] = stats_df[col].apply(lambda x: f'{x:.2%}')
fmt_table['Sharpe Ratio'] = stats_df['Sharpe Ratio'].apply(lambda x: f'{x:.2f}')

print("=== REGIME STATISTICS ===")
print(fmt_table[['Annual Return', 'Annual Volatility', 'Sharpe Ratio', 'Max Drawdown']].to_string())
print(f"\nComposite Score (State 0): {score_0}/3 signals won")
signal_summary = [
    f"  Return:  State 0 {stats_0['Annual Return']:.2%} vs State 1 {stats_1['Annual Return']:.2%} → {'State 0' if stats_0['Annual Return'] > stats_1['Annual Return'] else 'State 1'} wins",
    f"  Vol:     State 0 {stats_0['Annual Volatility']:.2%} vs State 1 {stats_1['Annual Volatility']:.2%} → {'State 0' if stats_0['Annual Volatility'] < stats_1['Annual Volatility'] else 'State 1'} wins",
    f"  Sharpe:  State 0 {stats_0['Sharpe Ratio']:.2f} vs State 1 {stats_1['Sharpe Ratio']:.2f} → {'State 0' if stats_0['Sharpe Ratio'] > stats_1['Sharpe Ratio'] else 'State 1'} wins",
]
for s in signal_summary:
    print(s)
print(f"\nLabel Assignment: State 0 → {label_0}, State 1 → {label_1}")

# Create regime_labels — canonical output variable for Phase 3 and Phase 4
regime_labels = regime_decoded.map({0: label_0, 1: label_1})
regime_labels.name = 'regime_label'

print(f"\nRegime labels distribution:")
print(regime_labels.value_counts())
print(f"\nOutput variable ready: regime_labels ({len(regime_labels):,} observations, {regime_labels.index[0].date()} to {regime_labels.index[-1].date()})")
```
