```python
def compute_skew(df: pd.DataFrame, min_dte: int = config.SKEW_MIN_DTE) -> pd.DataFrame:
    valid = df[(df["T_years"] > 0) & (df["iv"] > 0) & (df["oi"] > 0)].copy()
    valid["dte"] = valid["T_years"] * 365
    for expiry, grp in valid.groupby("expiry"):       # group by date string, NOT T_years
        dte = grp["dte"].iloc[0]
        puts = grp[grp["type"] == "put"]
        calls = grp[grp["type"] == "call"]
        put_idx = (puts["delta"] - config.SKEW_PUT_DELTA).abs().idxmin()
        call_idx = (calls["delta"] - config.SKEW_CALL_DELTA).abs().idxmin()
        put_iv = puts.loc[put_idx, "iv"] * 100
        call_iv = calls.loc[call_idx, "iv"] * 100
```

    compute_skew_25d(df, spot) behavior:
      - Filter: T_years > 0, iv > 0, oi > 0; add dte = T_years * 365
      - groupby("expiry") — NOT groupby("T_years")
      - Bucket assignment: dte <= 45 → front_month; 45 < dte <= 90 → second_month; else skip
      - Per bucket: take the first qualifying expiry (nearest DTE); skip if already filled
      - puts = grp where type=="put" AND strike < spot
      - calls = grp where type=="call" AND strike >= spot
      - if len(puts) < 2 or len(calls) < 2: set bucket to None and continue
      - put_idx = (puts["delta"] - config.SKEW_PUT_DELTA).abs().idxmin()  # SKEW_PUT_DELTA = -0.25
      - call_idx = (calls["delta"] - 0.25).abs().idxmin()  # 25Δ call — NOT config.SKEW_CALL_DELTA (that is 0.50)
      - put_iv = float(puts.loc[put_idx, "iv"] * 100)
      - call_iv = float(calls.loc[call_idx, "iv"] * 100)
      - bucket value: {"put_iv": put_iv, "call_iv": call_iv, "skew": put_iv - call_iv, "dte": float(dte)}
      - Return dict always has both keys: {"front_month": ..., "second_month": ...} (None if not found)

    compute_term_structure(df, spot) behavior:
      - Filter: T_years > 0, iv > 0, oi > 0; add dte = T_years * 365
      - groupby("expiry"), per group pick ATM: calls preferred, minimize |strike - spot|
        If no calls in group, fall back to puts
      - Collect points as list of {"dte": float, "atm_iv": float}, sorted ascending by dte
      - front_atm_iv = points[0]["atm_iv"] if len(points) >= 1 else None
      - back_atm_iv = points[-1]["atm_iv"] if len(points) >= 1 else None
      - Classification using points sorted by dte:
          flat: max_iv - min_iv < 1.0 * (max_dte - min_dte) / 30  → "flat"
            (less than 1 vol point per 30 DTE across full range)
          After flat check, compute total slope = (back_iv - front_iv) / (back_dte - front_dte) * 30:
            humped: middle third average > both front third average and back third average → "humped"
            inverted: slope < 0 (back_iv < front_iv) → "inverted"
            else → "normal"
          Edge case: len(points) < 2 → classification = "normal" (no slope to compute)
      - Return: {"points": [...], "classification": str, "front_atm_iv": float|None, "back_atm_iv": float|None}

    compute_rv20(spot_history: pd.Series) behavior:
      - spot_history must arrive sorted oldest-first (ascending date); function does NOT sort
      - if len(spot_history) < 21: return None
      - prices = spot_history.iloc[-21:].to_numpy(dtype=float)  # 21 prices = 20 returns
      - log_returns = np.log(prices[1:] / prices[:-1])
      - return float(np.sqrt(252) * log_returns.std(ddof=1))

    compute_vrp(iv30, rv20) behavior:
      - if iv30 is None or rv20 is None: return None
      - return iv30 - rv20
  </behavior>
