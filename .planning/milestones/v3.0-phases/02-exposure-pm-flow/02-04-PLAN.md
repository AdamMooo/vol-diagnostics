from __future__ import annotations

import pandas as pd
import pytest

from gex.exposure_engine import (
    compute_vex, compute_chex, strike_vex, strike_chex,
)
from gex.analytics import summarise
from gex.validation import _classify_vs_yesterday, load_yesterday


# ── Helpers ──────────────────────────────────────────────────────────────────

def _minimal_chain(n_calls: int = 1, n_puts: int = 1,
                   spot: float = 500.0) -> pd.DataFrame:
    """Single-strike chain for formula checks."""
    calls = pd.DataFrame({
        "strike": [spot] * n_calls,
        "type": ["call"] * n_calls,
        "oi": [100] * n_calls,
        "vanna": [0.05] * n_calls,
        "charm": [0.02] * n_calls,
        "gamma": [0.01] * n_calls,
    })
    puts = pd.DataFrame({
        "strike": [spot] * n_puts,
        "type": ["put"] * n_puts,
        "oi": [100] * n_puts,
        "vanna": [0.05] * n_puts,
        "charm": [0.02] * n_puts,
        "gamma": [0.01] * n_puts,
    })
    return pd.concat([calls, puts], ignore_index=True)


def _minimal_gex_df(net_gex: float = 1e9) -> pd.DataFrame:
    return pd.DataFrame({"strike": [500.0], "gex": [net_gex]})


def _minimal_profile_df() -> pd.DataFrame:
    return pd.DataFrame({
        "spot_level": [490.0, 495.0, 500.0, 505.0, 510.0],
        "net_gex":    [-1e9, -0.5e9, 0.0, 0.5e9, 1e9],
    })


# ── compute_vex ───────────────────────────────────────────────────────────────

def test_compute_vex_call_positive():
    df = _minimal_chain(n_calls=1, n_puts=0)
    result = compute_vex(df, spot=500.0)
    assert result["vex"].iloc[0] > 0

def test_compute_vex_put_negative():
    df = _minimal_chain(n_calls=0, n_puts=1)
    result = compute_vex(df, spot=500.0)
    assert result["vex"].iloc[0] < 0

def test_compute_vex_returns_copy():
    df = _minimal_chain()
    _ = compute_vex(df, spot=500.0)
    assert "vex" not in df.columns

def test_compute_vex_formula():
    df = pd.DataFrame({"strike": [500.0], "type": ["call"], "oi": [1], "vanna": [1.0], "charm": [0.0], "gamma": [0.0]})
    result = compute_vex(df, spot=500.0)
    expected = 1.0 * 1.0 * 1 * 100 * 500.0 * 0.01
    assert abs(result["vex"].iloc[0] - expected) < 1e-9

def test_compute_vex_spot_not_squared():
    # Verify spot*0.01, not spot**2*0.01
    df = pd.DataFrame({"strike": [500.0], "type": ["call"], "oi": [1], "vanna": [1.0], "charm": [0.0], "gamma": [0.0]})
    result = compute_vex(df, spot=500.0)
    wrong = 1.0 * 1 * 100 * 500.0**2 * 0.01
    assert abs(result["vex"].iloc[0]) != abs(wrong)


# ── compute_chex ──────────────────────────────────────────────────────────────

def test_compute_chex_call_positive():
    df = _minimal_chain(n_calls=1, n_puts=0)
    result = compute_chex(df, spot=500.0)
    assert result["chex"].iloc[0] > 0  # positive charm, call sign +1

def test_compute_chex_put_negative():
    df = _minimal_chain(n_calls=0, n_puts=1)
    result = compute_chex(df, spot=500.0)
    assert result["chex"].iloc[0] < 0

def test_compute_chex_returns_copy():
    df = _minimal_chain()
    _ = compute_chex(df, spot=500.0)
    assert "chex" not in df.columns

def test_compute_chex_formula():
    df = pd.DataFrame({"strike": [500.0], "type": ["call"], "oi": [1], "charm": [1.0], "vanna": [0.0], "gamma": [0.0]})
    result = compute_chex(df, spot=500.0)
    expected = 1.0 * 1.0 * 1 * 100 * 500.0 * 0.01
    assert abs(result["chex"].iloc[0] - expected) < 1e-9


# ── strike_vex / strike_chex ──────────────────────────────────────────────────

def test_strike_vex_aggregates_by_strike():
    df = pd.DataFrame({
        "strike": [490.0, 490.0, 500.0],
        "type":   ["call", "put", "call"],
        "oi":     [100, 100, 50],
        "vanna":  [0.05, 0.05, 0.05],
        "charm":  [0.0, 0.0, 0.0],
        "gamma":  [0.0, 0.0, 0.0],
    })
    df_vex = compute_vex(df, spot=500.0)
    result = strike_vex(df_vex)
    assert list(result.columns) == ["strike", "vex"]
    assert result["strike"].is_monotonic_increasing
    # 490 strike has one call and one put — they partially offset
    row_490 = result[result["strike"] == 490.0]["vex"].iloc[0]
    assert row_490 == 0.0  # equal oi, opposite sign

def test_strike_chex_columns():
    df = _minimal_chain()
    df_chex = compute_chex(df, spot=500.0)
    result = strike_chex(df_chex)
    assert list(result.columns) == ["strike", "chex"]


# ── summarise() extension ─────────────────────────────────────────────────────

def test_summarise_backward_compat():
    s_df = _minimal_gex_df(1e9)
    p_df = _minimal_profile_df()
    r = summarise(s_df, p_df, spot=500.0)
    assert r["net_vex"] is None
    assert r["net_chex"] is None
    assert r["delta_hedge_flow"] is None
    assert "net_gex" in r
    assert "gamma_regime" in r

def test_summarise_with_kwargs():
    s_df = _minimal_gex_df(1e9)
    p_df = _minimal_profile_df()
    r = summarise(s_df, p_df, spot=500.0, net_vex=2e9, net_chex=-1e9, delta_hedge_flow=8.4e9)
    assert r["net_vex"] == 2e9
    assert r["net_chex"] == -1e9
    assert r["delta_hedge_flow"] == 8.4e9


# ── _classify_vs_yesterday ────────────────────────────────────────────────────

def _prior(regime: str, net_gex: float) -> pd.Series:
    return pd.Series({"gamma_regime": regime, "net_gex": net_gex})

def test_classify_flipped_pos_to_neg():
    assert _classify_vs_yesterday(1e9, "negative", _prior("positive", 1e9)) == "FLIPPED"

def test_classify_flipped_neg_to_pos():
    assert _classify_vs_yesterday(1e9, "positive", _prior("negative", -1e9)) == "FLIPPED"

def test_classify_flipped_neutral_to_pos():
    assert _classify_vs_yesterday(1e9, "positive", _prior("neutral", 0.1e9)) == "FLIPPED"

def test_classify_intensified():
    assert _classify_vs_yesterday(1.2e9, "positive", _prior("positive", 1e9)) == "INTENSIFIED"

def test_classify_eased():
    assert _classify_vs_yesterday(0.8e9, "positive", _prior("positive", 1e9)) == "EASED"

def test_classify_unchanged_within_band():
    assert _classify_vs_yesterday(1.03e9, "positive", _prior("positive", 1e9)) == "UNCHANGED"

def test_classify_zero_prior_gex():
    assert _classify_vs_yesterday(1e9, "positive", _prior("positive", 0.0)) == "UNCHANGED"


# ── load_yesterday absent store ───────────────────────────────────────────────

def test_load_yesterday_returns_none_when_store_absent(monkeypatch, tmp_path):
    import gex.validation as val
    monkeypatch.setattr(val, "STORE", tmp_path / "nonexistent.parquet")
    result = load_yesterday("SPY")
    assert result is None
